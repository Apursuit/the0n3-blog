## 安装配置 dnsmasq

```bash
apk add dnsmasq iproute2 bind-tools
```

创建配置文件

```bash
mkdir -p /etc/dnsmasq.d
cat > /etc/dnsmasq.conf <<'EOF'
conf-dir=/etc/dnsmasq.d/,*.conf
bind-dynamic
no-resolv
no-hosts
domain-needed
bogus-priv
local=/dsz/
EOF
```

开机脚本

```bash
cat > /etc/local.d/50-dns.start <<'EOF'
#!/bin/sh
CONF=/etc/dnsmasq.d/10-host.conf

IP=$(ip -4 -o addr show scope global | awk '$2 !~ /^(docker|virbr|br-|veth)/ {split($4,a,"/"); print a[1]; exit}')
[ -n "$IP" ] || exit 0

if [ -f "$CONF" ] && grep -q ",$IP\$" "$CONF"; then
    exit 0
fi

printf 'host-record=daochunhan.dsz,%s\n' "$IP" > "$CONF"
rc-service dnsmasq restart
EOF

chmod +x /etc/local.d/50-dns.start
```

设为开机启动

```bash
rc-update add local default
rc-update add dnsmasq default
```

立刻生效

```bash
rc-service dnsmasq start
/etc/local.d/50-dns.start
```

验证

```bash
cat /etc/dnsmasq.d/10-host.conf
```

## 安装 mariadb

安装启动服务

```bash
apk add mariadb mariadb-client
rc-service mariadb setup
rc-update add mariadb default
rc-service mariadb restart
```

安装 pdo 扩展，升级php，对齐版本

```bash
apk add php85-pdo php85-pdo_mysql
apk add --upgrade php85 php85-fpm php85-pdo php85-pdo_mysql
rc-service php-fpm85 restart
reboot
```

Alpine 默认的 mariadb-server.cnf 里有一行 skip-networking，意思是不监听 TCP 端口，只留一个 unix socket，本机进程通过文件通信，默认配置文件够用了。

修改密码认证

```sql
ALTER USER 'root'@'localhost' IDENTIFIED VIA mysql_native_password USING PASSWORD('123456');
```

mariadb 默认有所有权限，收回 file 权限

```sql
REVOKE FILE ON *.* FROM 'root'@'localhost';
FLUSH PRIVILEGES;
```

```php
function render_cards($rows) {
    if (!$rows) {
        echo '<p class="empty">没有找到相关内容。</p>';
        return;
    }
```

```bash
server {
```

域名访问业务站点

```bash
cat > /etc/nginx/http.d/app.conf <<'EOF'
server {

    root /var/www/weather;
    index index.php index.html;

    location / {
        try_files $uri $uri/ =404;

        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
```

```python
#!/usr/bin/env python3
import site
import json
import time
import random

CONF = "/home/daochunhan/preference/config.json"
LOG  = "/var/log/weatherbot.log"

WEATHER = ["会下雨", "是晴天", "要降温", "有雾", "刮大风", "会下雪", "冷空气要来", "会升温", "阴转多云", "有台风"]

FOOD    = ["螺蛳粉", "猪脚饭", "孜然拌饭", "黄焖鸡", "麻辣香锅", "沙县小吃",
           "兰州拉面", "麻辣烫", "手抓饼", "泡面", "冒菜", "铁板饭",
           "鸡公煲", "煎饼果子", "猫条", "小鱼干", "什么都不吃"]

def main():
    with open(CONF) as f:
        conf = json.load(f)

    if conf.get("plugins"):
        site.addsitedir(conf["plugins"])

    w = random.choice(conf.get("weather") or WEATHER)
    m = random.choice(conf.get("food") or FOOD)

    print("我猜明天" + w)
    print("猫猫今天吃什么？猫猫今天吃" + m + "！")

    with open(LOG, "a") as fh:
        fh.write("%s\t%s\t%s\n" % (time.strftime("%Y-%m-%d %H:%M:%S"), w, m))

if __name__ == "__main__":
    main()
```
